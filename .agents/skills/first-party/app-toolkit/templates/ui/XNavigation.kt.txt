package <base>.feature.x.ui.navigation

import androidx.compose.ui.res.stringResource
import androidx.navigation3.runtime.NavKey
import <base>.feature.x.R
import <base>.feature.x.ui.XScreen
import com.mihaicristiancondrea.android.libs.apptoolkit.navigation.graph.PaneRole
import com.mihaicristiancondrea.android.libs.apptoolkit.navigation.graph.ShellGraphBuilder
import kotlinx.serialization.Serializable

/** Serializable key for the inventory catalog destination. */
@Serializable
data object XScreen : NavKey

/**
 * Registers the catalog destination unless the shell already defines [XScreen].
 *
 * @param onOpenDetails Receives the selected item's key for host-owned navigation.
 */
fun ShellGraphBuilder.xScreen(onOpenDetails: (String) -> Unit) {
    pageIfAbsent<XScreen>(paneRole = PaneRole.None, title = { stringResource(id = R.string.x_title) }) {
        XScreen(onOpenDetails = onOpenDetails)
    }
}
