package <base>.feature.x.ui.views

import androidx.compose.foundation.clickable
import androidx.compose.material3.Text
import androidx.compose.runtime.Composable
import androidx.compose.ui.Modifier
import <base>.feature.x.ui.constants.XConstants
import <base>.feature.x.ui.models.XItem

/** Displays a catalog title and makes the whole row a detail action. */
@Composable
internal fun XView(item: XItem, onClick: () -> Unit, modifier: Modifier = Modifier) {
    Text(
        text = item.title.asString(),
        maxLines = XConstants.MAX_TITLE_LINES,
        modifier = modifier.clickable(onClick = onClick),
    )
}
