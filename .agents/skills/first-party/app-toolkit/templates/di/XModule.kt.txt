package <base>.feature.x.di

import <base>.feature.x.data.remote.XRemoteDataSource
import <base>.feature.x.data.repositories.DefaultXRepository
import <base>.feature.x.data.repositories.XRepository
import <base>.feature.x.ui.XViewModel
import org.koin.core.module.Module
import org.koin.core.module.dsl.viewModel
import org.koin.core.qualifier.named
import org.koin.dsl.module

/**
 * Registers the catalog sources, repository and ViewModel.
 *
 * The host supplies a JSON-enabled HTTP client, telemetry and a String qualified by `xEndpoint`.
 */
val xModule: Module = module {
    single {
        XRemoteDataSource(
            client = get(),
            endpoint = get(qualifier = named(name = "xEndpoint")),
        )
    }
    single<XRepository> { DefaultXRepository(remote = get()) }
    viewModel { XViewModel(repository = get(), telemetryRepository = get()) }
}
