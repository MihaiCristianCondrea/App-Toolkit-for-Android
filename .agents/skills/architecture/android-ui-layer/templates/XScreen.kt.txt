private const val X_SCREEN_NAME = "X"
private const val X_SCREEN_CLASS = "XScreen"

/**
 * The X screen. Owns the [XViewModel], tracking, messages and navigation, and hands the rendering
 * to [XScreenContent].
 *
 * Tracking reports through `LocalTelemetry`, which `ShellHost` provides. To log an event of its
 * own, the screen reads `val telemetry = LocalTelemetry.current` and calls it from a callback.
 */
@Composable
fun XScreen() {
    val viewModel: XViewModel = koinViewModel()
    val state: XUiState by viewModel.state.collectAsStateWithLifecycle()
    val navigator = LocalShellNavigator.current

    TrackScreenView(
        screenName = X_SCREEN_NAME,
        screenClass = X_SCREEN_CLASS,
    )

    TrackScreenState(
        screenName = X_SCREEN_NAME,
        state = state.items,
    )

    XScreenContent(
        state = state,
        onEvent = viewModel::onEvent,
        onOpenDetails = { id -> navigator.navigate(XDetailsRoute(id)) },
        contentPadding = contentPadding(),
    )

    MessageHost(viewModel = viewModel)
}

/**
 * Renders [state] and reports what the user does through callbacks. Holds no ViewModel, injects
 * nothing and does not navigate, so it renders in a preview or a test with plain values.
 *
 * @param onEvent Receives the events [XViewModel] handles.
 * @param onOpenDetails Opens an entry's details. Navigation belongs to the caller.
 * @param contentPadding Padding from the shell, applied inside the list and the state screens.
 */
@Composable
internal fun XScreenContent(
    state: XUiState,
    onEvent: (XEvent) -> Unit,
    onOpenDetails: (id: String) -> Unit,
    modifier: Modifier = Modifier,
    contentPadding: PaddingValues = PaddingValues(),
) {
    ScreenStateHandler(
        state = state.items,
        modifier = modifier,
        contentPadding = contentPadding,
        onRetry = { onEvent(XEvent.Load) },
    ) { ready ->
        LazyColumn(contentPadding = contentPadding) {
            items(items = ready.value, key = { it.key }) { item ->
                XRow(item = item, onClick = { onOpenDetails(item.key) })
            }
        }
    }
}
